alias c='clear'  clears terminal

alias ll='ls -lah'  lists files with details

alias ..='cd ..' moves one directory up

alias home='cd ~' goes to home directory

alias gs='git status' checks git status

alias ga='git add .'  stages all changes

alias gc='git commit' creates a commit

alias gp='git push'  pushes changes

alias gl='git log --oneline'  displays commit history
